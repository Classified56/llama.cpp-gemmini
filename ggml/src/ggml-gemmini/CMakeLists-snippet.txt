# Add these sources to ggml/src/ggml-gemmini/CMakeLists.txt.

ggml_add_backend_library(ggml-gemmini
    ggml-gemmini.cpp
    ggml-gemmini-runtime.cpp
    ggml-gemmini-kernel.cpp
)

target_include_directories(ggml-gemmini PRIVATE
    ${CMAKE_CURRENT_SOURCE_DIR}
)

# Existing CPU-buffer linkage remains appropriate for a coherent,
# CPU-attached accelerator.
if (TARGET ggml-cpu)
    target_link_libraries(ggml-gemmini PRIVATE ggml-cpu)
endif()

set(GEMMINI_INCLUDE_DIR "" CACHE PATH
    "Path containing gemmini.h and generated Gemmini headers")

option(GGML_GEMMINI_ENABLE_RUNTIME
    "Compile the Gemmini hardware runtime wrapper"
    OFF)

option(GGML_GEMMINI_RUNTIME_FORCE_AVAILABLE
    "Expose the runtime without hardware for registration tests"
    OFF)

if (GGML_GEMMINI_RUNTIME_FORCE_AVAILABLE)
    target_compile_definitions(ggml-gemmini PRIVATE
        GGML_GEMMINI_RUNTIME_FORCE_AVAILABLE)
endif()

if (GGML_GEMMINI_ENABLE_RUNTIME)
    if (NOT GEMMINI_INCLUDE_DIR)
        message(FATAL_ERROR
            "GGML_GEMMINI_ENABLE_RUNTIME requires GEMMINI_INCLUDE_DIR")
    endif()

    target_include_directories(ggml-gemmini PRIVATE
        ${GEMMINI_INCLUDE_DIR})

    target_compile_definitions(ggml-gemmini PRIVATE
        GGML_GEMMINI_RUNTIME_HAS_HARDWARE)
endif()
